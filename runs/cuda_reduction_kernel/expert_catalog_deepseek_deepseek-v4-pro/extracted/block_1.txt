// reduce.cu
// CUDA reduction: sum of a float array.
// Compile with: nvcc -std=c++11 -o reduce_test reduce.cu
// Run: ./reduce_test

#include <cuda_runtime.h>
#include <cstdio>
#include <cmath>
#include <stdexcept>
#include <string>
#include <memory>
#include <vector>
#include <algorithm>
#include <numeric>
#include <iostream>
#include <iomanip>

// -----------------------------------------------------------------------------
// Error handling helpers
// -----------------------------------------------------------------------------
#define CUDA_CHECK(call)                                                       \
  do {                                                                         \
    cudaError_t err = call;                                                    \
    if (err != cudaSuccess) {                                                  \
      throw std::runtime_error(std::string("CUDA error at ") + __FILE__ +      \
                               ":" + std::to_string(__LINE__) + " " +          \
                               cudaGetErrorString(err));                       \
    }                                                                          \
  } while (0)

#define KERNEL_CHECK()                                                         \
  do {                                                                         \
    cudaError_t err = cudaGetLastError();                                      \
    if (err != cudaSuccess) {                                                  \
      throw std::runtime_error(std::string("Kernel launch error at ") +        \
                               __FILE__ + ":" + std::to_string(__LINE__) +     \
                               " " + cudaGetErrorString(err));                 \
    }                                                                          \
    CUDA_CHECK(cudaDeviceSynchronize());                                       \
  } while (0)

// -----------------------------------------------------------------------------
// RAII wrapper for device memory
// -----------------------------------------------------------------------------
struct DeviceMemory {
  float* ptr = nullptr;
  size_t size = 0;

  DeviceMemory() = default;
  explicit DeviceMemory(size_t bytes) : size(bytes) {
    CUDA_CHECK(cudaMalloc(&ptr, bytes));
  }
  ~DeviceMemory() {
    if (ptr) cudaFree(ptr);
  }
  // Non-copyable, movable
  DeviceMemory(const DeviceMemory&) = delete;
  DeviceMemory& operator=(const DeviceMemory&) = delete;
  DeviceMemory(DeviceMemory&& other) noexcept : ptr(other.ptr), size(other.size) {
    other.ptr = nullptr;
    other.size = 0;
  }
  DeviceMemory& operator=(DeviceMemory&& other) noexcept {
    if (this != &other) {
      if (ptr) cudaFree(ptr);
      ptr = other.ptr;
      size = other.size;
      other.ptr = nullptr;
      other.size = 0;
    }
    return *this;
  }
};

// -----------------------------------------------------------------------------
// Reduction kernel
// -----------------------------------------------------------------------------
// Each block reduces its assigned segment using a grid‑stride loop,
// stores the partial sum in shared memory, performs a tree reduction
// down to 32 elements, and finally uses warp shuffles to obtain the
// block’s total. The result is written to output[blockIdx.x].
//
// Assumptions:
//   - blockDim.x >= 32 (enforced by the host).
//   - The kernel is launched with enough dynamic shared memory:
//     blockDim.x * sizeof(float).
//   - input and output do not overlap.
__global__ void reduce_kernel(const float* input, float* output, size_t n) {
  extern __shared__ float sdata[];

  const size_t tid = threadIdx.x;
  const size_t global_tid = blockIdx.x * blockDim.x + tid;
  const size_t stride = gridDim.x * blockDim.x;

  // 1. Grid‑stride accumulation – each thread sums its own elements.
  float sum = 0.0f;
  for (size_t i = global_tid; i < n; i += stride) {
    sum += input[i];
  }
  sdata[tid] = sum;
  __syncthreads();  // all threads have written to shared memory

  // 2. Tree reduction in shared memory (down to 32 elements).
  for (size_t s = blockDim.x / 2; s > 32; s >>= 1) {
    if (tid < s) {
      sdata[tid] += sdata[tid + s];
    }
    __syncthreads();  // ensure all reads of the previous level are done
  }

  // 3. Warp‑level reduction using shuffle instructions.
  //    Only the first warp (threads 0..31) participates.
  if (tid < 32) {
    float warp_sum = sdata[tid];
    // __shfl_down_sync with full mask – safe because blockDim.x >= 32.
    warp_sum += __shfl_down_sync(0xffffffff, warp_sum, 16);
    warp_sum += __shfl_down_sync(0xffffffff, warp_sum, 8);
    warp_sum += __shfl_down_sync(0xffffffff, warp_sum, 4);
    warp_sum += __shfl_down_sync(0xffffffff, warp_sum, 2);
    warp_sum += __shfl_down_sync(0xffffffff, warp_sum, 1);
    if (tid == 0) {
      output[blockIdx.x] = warp_sum;
    }
  }
}

// -----------------------------------------------------------------------------
// Public host API
// -----------------------------------------------------------------------------
/// Sums the elements of `host_input` (length `n`) on the GPU.
/// Returns the sum as a float.
/// Throws std::runtime_error on any CUDA error (allocation, copy, kernel).
/// Handles n == 0 and n == 1 correctly.
float gpu_reduce_sum(const float* host_input, size_t n) {
  if (n == 0) return 0.0f;

  // Query device properties to respect grid size limits.
  int device;
  CUDA_CHECK(cudaGetDevice(&device));
  cudaDeviceProp prop;
  CUDA_CHECK(cudaGetDeviceProperties(&prop, device));

  const int blockSize = 256;               // must be >= 32
  const int maxGridSize = prop.maxGridSize[0];

  // Choose a grid size that covers all elements without exceeding the limit.
  // The grid‑stride loop inside the kernel handles any remainder.
  int gridSize = static_cast<int>(std::min(
      static_cast<size_t>(maxGridSize),
      (n + blockSize - 1) / blockSize));

  // Allocate device input and copy data.
  DeviceMemory d_input(n * sizeof(float));
  CUDA_CHECK(cudaMemcpy(d_input.ptr, host_input, n * sizeof(float),
                        cudaMemcpyHostToDevice));

  // Allocate partial sums (one float per block).
  DeviceMemory d_partial(gridSize * sizeof(float));

  // First reduction pass: each block writes its partial sum.
  size_t sharedBytes = blockSize * sizeof(float);
  reduce_kernel<<<gridSize, blockSize, sharedBytes>>>(
      d_input.ptr, d_partial.ptr, n);
  KERNEL_CHECK();

  // If only one block was used, the partial sum is already the final result.
  if (gridSize == 1) {
    float result;
    CUDA_CHECK(cudaMemcpy(&result, d_partial.ptr, sizeof(float),
                          cudaMemcpyDeviceToHost));
    return result;
  }

  // Second pass: reduce the partial sums array (length = gridSize) with a
  // single block. The same kernel is reused; output is a single float.
  DeviceMemory d_final(sizeof(float));
  reduce_kernel<<<1, blockSize, sharedBytes>>>(
      d_partial.ptr, d_final.ptr, static_cast<size_t>(gridSize));
  KERNEL_CHECK();

  float result;
  CUDA_CHECK(cudaMemcpy(&result, d_final.ptr, sizeof(float),
                        cudaMemcpyDeviceToHost));
  return result;
}

// -----------------------------------------------------------------------------
// CPU reference (sequential sum in double precision for accuracy)
// -----------------------------------------------------------------------------
float cpu_reduce_sum(const float* data, size_t n) {
  double sum = 0.0;
  for (size_t i = 0; i < n; ++i) sum += data[i];
  return static_cast<float>(sum);
}

// -----------------------------------------------------------------------------
// Test harness
// -----------------------------------------------------------------------------
bool approx_equal(float a, float b, float tol = 1e-5f) {
  float denom = std::max(std::abs(b), 1e-6f);
  return std::abs(a - b) / denom < tol;
}

void run_test(const std::string& name, const std::vector<float>& data) {
  float cpu = cpu_reduce_sum(data.data(), data.size());
  float gpu = gpu_reduce_sum(data.data(), data.size());
  bool pass = approx_equal(cpu, gpu);
  std::cout << std::left << std::setw(30) << name
            << " CPU: " << std::setw(12) << cpu
            << " GPU: " << std::setw(12) << gpu
            << (pass ? "PASS" : "FAIL") << "\n";
  if (!pass) {
    std::cerr << "  Difference: " << std::abs(cpu - gpu) << "\n";
  }
}

int main() {
  try {
    std::cout << "CUDA Reduction Tests\n";
    std::cout << "====================\n";

    // n = 0
    run_test("n=0", {});

    // n = 1
    run_test("n=1", {42.0f});

    // Small array
    run_test("small (n=5)", {1.0f, 2.0f, 3.0f, 4.0f, 5.0f});

    // Non‑power‑of‑two size
    std::vector<float> np2(1000);
    std::iota(np2.begin(), np2.end(), 1.0f);
    run_test("non-power-of-two (n=1000)", np2);

    // Large array (1M elements)
    std::vector<float> large(1'000'000);
    std::generate(large.begin(), large.end(),
                  [n = 0]() mutable { return static_cast<float>(n++); });
    run_test("large (n=1e6)", large);

    // Negative values
    std::vector<float> neg(1000);
    std::generate(neg.begin(), neg.end(),
                  [n = 0]() mutable { return (n++ % 2 == 0) ? 1.0f : -1.0f; });
    run_test("negative values (n=1000)", neg);

    // Mixed magnitudes (large + small)
    std::vector<float> mixed(1000, 1e-10f);
    mixed[0] = 1e10f;
    run_test("mixed magnitudes (n=1000)", mixed);

    std::cout << "\nAll tests completed.\n";
  } catch (const std::exception& e) {
    std::cerr << "Fatal error: " << e.what() << "\n";
    return 1;
  }
  return 0;
}
